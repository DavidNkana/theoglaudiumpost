    /**
     * Parse tool calls from MiniMax M2 XML format
     * Format: <minimax:tool_call><invoke name="..."><parameter name="...">value</parameter></invoke></minimax:tool_call>
     */
    private List<Map<String, Object>> parseToolCalls(String content) {
        List<Map<String, Object>> toolCalls = new ArrayList<>();

        if (content == null || !content.contains("<minimax:tool_call>")) {
            return toolCalls;
        }

        Pattern toolCallPattern = Pattern.compile("<minimax:tool_call>(.*?)</minimax:tool_call>", Pattern.DOTALL);
        Pattern invokePattern = Pattern.compile("<invoke name=\"([^\"]+)\">(.*?)</invoke>", Pattern.DOTALL);
        Pattern paramPattern = Pattern.compile("<parameter name=\"([^\"]+)\">([^<]*)</parameter>", Pattern.DOTALL);

        Matcher toolCallMatcher = toolCallPattern.matcher(content);
        while (toolCallMatcher.find()) {
            String toolCallContent = toolCallMatcher.group(1);

            Matcher invokeMatcher = invokePattern.matcher(toolCallContent);
            while (invokeMatcher.find()) {
                String functionName = invokeMatcher.group(1);
                String invokeContent = invokeMatcher.group(2);

                Map<String, Object> arguments = new LinkedHashMap<>();
                Matcher paramMatcher = paramPattern.matcher(invokeContent);
                while (paramMatcher.find()) {
                    String paramName = paramMatcher.group(1);
                    String paramValue = paramMatcher.group(2).trim();

                    // Try to parse as JSON array if it looks like one
                    if (paramValue.startsWith("[") && paramValue.endsWith("]")) {
                        try {
                            JsonArray arr = JsonParser.parseString(paramValue).getAsJsonArray();
                            List<String> list = new ArrayList<>();
                            for (JsonElement el : arr) {
                                list.add(el.getAsString());
                            }
                            arguments.put(paramName, list);
                        } catch (Exception e) {
                            arguments.put(paramName, paramValue);
                        }
                    } else {
                        arguments.put(paramName, paramValue);
                    }
                }

                Map<String, Object> toolCall = new LinkedHashMap<>();
                toolCall.put("name", functionName);
                toolCall.put("arguments", arguments);
                toolCalls.add(toolCall);
            }
        }

        return toolCalls;
    }

    /**
     * Extract thinking content from response (preserves interleaved thinking)
     */
    private String extractThinking(String content) {
        if (content == null) return "";

        Pattern thinkPattern = Pattern.compile("<think>(.*?)</think>", Pattern.DOTALL);
        Matcher matcher = thinkPattern.matcher(content);
        StringBuilder thinking = new StringBuilder();
        while (matcher.find()) {
            if (thinking.length() > 0) thinking.append("\n");
            thinking.append(matcher.group(1).trim());
        }
        return thinking.toString();
    }

    /**
     * Remove thinking tags from content for final output
     */
    private String removeThinking(String content) {
        if (content == null) return "";
        return content.replaceAll("<think>.*?</think>", "").trim();
    }

    /**
     * Parse an argument value that might be a List or a JSON String representation of an array.
     * This handles both cases since Gson may deserialize array arguments as either type.
     *
     * @ModifiedByJunie 2025-12-10
     */
    @SuppressWarnings("unchecked")
    private List<String> parseListArgument(Object value) {
        if (value == null) {
            return List.of();
        }
        if (value instanceof List) {
            return (List<String>) value;
        }
        if (value instanceof String) {
            String strValue = (String) value;
            // Try to parse as JSON array
            if (strValue.startsWith("[") && strValue.endsWith("]")) {
                try {
                    JsonArray arr = JsonParser.parseString(strValue).getAsJsonArray();
                    List<String> list = new ArrayList<>();
                    for (JsonElement el : arr) {
                        list.add(el.getAsString());
                    }
                    return list;
                } catch (Exception e) {
                    // Not a valid JSON array, treat as single value
                    return List.of(strValue);
                }
            }
            // Single string value
            return List.of(strValue);
        }
        return List.of();
    }

    /**
     * Execute tool call and return result
     */
    private String executeToolCall(Map<String, Object> toolCall) {
        String name = (String) toolCall.get("name");
        @SuppressWarnings("unchecked")
        Map<String, Object> arguments = (Map<String, Object>) toolCall.get("arguments");

        return switch (name) {
            case "search_web" -> executeWebSearch(arguments);
            case "scrape_webpage" -> executeWebScrape(arguments);
            default -> gson.toJson(Map.of("error", "Unknown tool: " + name));
        };
    }

    /**
     * Make a chat completion request with optional tool support
     */
    public Map<String, Object> chatCompletion(List<Map<String, Object>> messages, List<Map<String, Object>> tools, boolean toolChoice) throws IOException {
        if (apiKey == null || apiKey.isEmpty()) {
            throw new IOException("MiniMax API key not configured. Set minimax.api.key in application.properties");
        }

        Map<String, Object> requestBody = new LinkedHashMap<>();
        requestBody.put("model", model);
        requestBody.put("messages", messages);
        requestBody.put("temperature", 1.0); // Recommended for MiniMax M2

        if (tools != null && !tools.isEmpty()) {
            requestBody.put("tools", tools);
            requestBody.put("tool_choice", toolChoice ? "auto" : "none");
        }

        try {
            HttpHeaders headers = createHeaders();
            HttpEntity<String> entity = new HttpEntity<>(gson.toJson(requestBody), headers);

            logger.info("Making MiniMax API request to: " + getChatCompletionsUrl());
            logger.fine("Request body: " + gson.toJson(requestBody));

            ResponseEntity<String> response = restTemplate.exchange(
                    getChatCompletionsUrl(),
                    HttpMethod.POST,
                    entity,
                    String.class
            );

            if (response.getStatusCode().is2xxSuccessful() && response.getBody() != null) {
                JsonObject responseJson = JsonParser.parseString(response.getBody()).getAsJsonObject();

                // Extract the assistant's message content
                JsonArray choices = responseJson.getAsJsonArray("choices");
                if (choices != null && choices.size() > 0) {
                    JsonObject firstChoice = choices.get(0).getAsJsonObject();
                    JsonObject message = firstChoice.getAsJsonObject("message");

                    String content = message.has("content") && !message.get("content").isJsonNull()
                            ? message.get("content").getAsString()
                            : "";

                    // Check for tool calls in the response
                    List<Map<String, Object>> parsedToolCalls = new ArrayList<>();

                    // First check OpenAI-style tool_calls
                    if (message.has("tool_calls") && message.get("tool_calls").isJsonArray()) {
                        JsonArray toolCallsArray = message.getAsJsonArray("tool_calls");
                        for (JsonElement tc : toolCallsArray) {
                            JsonObject tcObj = tc.getAsJsonObject();
                            JsonObject fn = tcObj.getAsJsonObject("function");
                            Map<String, Object> toolCall = new LinkedHashMap<>();
                            toolCall.put("id", tcObj.has("id") ? tcObj.get("id").getAsString() : UUID.randomUUID().toString());
                            toolCall.put("name", fn.get("name").getAsString());
                            toolCall.put("arguments", gson.fromJson(fn.get("arguments").getAsString(), Map.class));
                            parsedToolCalls.add(toolCall);
                        }
                    }

                    // Also check for XML-style tool calls in content (MiniMax native format)
                    if (content.contains("<minimax:tool_call>")) {
                        parsedToolCalls.addAll(parseToolCalls(content));
                    }

                    Map<String, Object> result = new LinkedHashMap<>();
                    result.put("content", content);
                    result.put("thinking", extractThinking(content));
                    result.put("tool_calls", parsedToolCalls);
                    result.put("finish_reason", firstChoice.has("finish_reason")
                            ? firstChoice.get("finish_reason").getAsString()
                            : "stop");

                    return result;
                }

                throw new IOException("No choices in MiniMax API response");
            } else {
                throw new IOException("MiniMax API error: " + response.getStatusCode());
            }
        } catch (Exception e) {
            logger.log(Level.SEVERE, "MiniMax API error: " + e.getMessage(), e);
            throw new IOException("Error calling MiniMax API: " + e.getMessage(), e);
        }
    }

    /**
     * Chat with tool calling loop - handles multiple rounds of tool execution
     * Uses OpenAI-compatible format for tool calls and results since we're using the OpenAI-compatible endpoint.
     * Each tool call gets an ID and each tool result references that ID via tool_call_id.
     *
     * @ModifiedByJunie 2025-12-10
     */
    public String chatWithTools(String systemPrompt, String userPrompt, int maxToolRounds) throws IOException {
        List<Map<String, Object>> messages = new ArrayList<>();

        // Add system message
        messages.add(Map.of("role", "system", "content", systemPrompt));

        // Add user message
        messages.add(Map.of("role", "user", "content", userPrompt));

        List<Map<String, Object>> tools = getAllTools();

        for (int round = 0; round < maxToolRounds; round++) {
            Map<String, Object> response = chatCompletion(messages, tools, true);

            String content = (String) response.get("content");
            @SuppressWarnings("unchecked")
            List<Map<String, Object>> toolCalls = (List<Map<String, Object>>) response.get("tool_calls");

            // If no tool calls, add simple assistant message and return
            if (toolCalls == null || toolCalls.isEmpty()) {
                Map<String, Object> assistantMessage = new LinkedHashMap<>();
                assistantMessage.put("role", "assistant");
                assistantMessage.put("content", content != null ? content : "");
                messages.add(assistantMessage);
                return removeThinking(content != null ? content : "");
            }

            // Build assistant message with tool_calls in OpenAI format
            Map<String, Object> assistantMessage = new LinkedHashMap<>();
            assistantMessage.put("role", "assistant");
            assistantMessage.put("content", content != null ? content : "");

            // Format tool_calls and store IDs for matching with results
            List<Map<String, Object>> formattedToolCalls = new ArrayList<>();
            List<String> toolCallIds = new ArrayList<>();

            for (int i = 0; i < toolCalls.size(); i++) {
                Map<String, Object> toolCall = toolCalls.get(i);
                // Generate a consistent ID format: call_function_<timestamp>_<index>
                String toolCallId = toolCall.get("id") != null && !((String) toolCall.get("id")).isEmpty()
                        ? (String) toolCall.get("id")
                        : "call_function_" + System.currentTimeMillis() + "_" + i;
                String functionName = (String) toolCall.get("name");
                @SuppressWarnings("unchecked")
                Map<String, Object> arguments = (Map<String, Object>) toolCall.get("arguments");

                Map<String, Object> formattedCall = new LinkedHashMap<>();
                formattedCall.put("id", toolCallId);
                formattedCall.put("type", "function");
                formattedCall.put("function", Map.of(
                        "name", functionName,
                        "arguments", arguments != null ? gson.toJson(arguments) : "{}"
                ));
                formattedToolCalls.add(formattedCall);
                toolCallIds.add(toolCallId);

                // Store the ID back in the toolCall for later use
                toolCall.put("id", toolCallId);
            }
            assistantMessage.put("tool_calls", formattedToolCalls);
            messages.add(assistantMessage);

            // Execute each tool call and add result with OpenAI format (separate message per result)
            for (int i = 0; i < toolCalls.size(); i++) {
                Map<String, Object> toolCall = toolCalls.get(i);
                String result = executeToolCall(toolCall);
                String toolCallId = toolCallIds.get(i);

                // Add tool result message in OpenAI format with tool_call_id
                Map<String, Object> toolMessage = new LinkedHashMap<>();
                toolMessage.put("role", "tool");
                toolMessage.put("tool_call_id", toolCallId);
                toolMessage.put("content", result);
                messages.add(toolMessage);
            }
        }

        // If we've exhausted tool rounds, get final response without tools
        Map<String, Object> finalResponse = chatCompletion(messages, null, false);
        return removeThinking((String) finalResponse.get("content"));
    }

    /**
     * Extract JSON object from response text, handling various formats the model might return.
     * Looks for JSON object starting with { and ending with }
     *
     * @ModifiedByJunie 2025-12-10
     */
    private String extractJsonObject(String text) {
        if (text == null || text.isEmpty()) {
            return "{}";
        }

        // Remove markdown code fences
        String cleaned = text
                .replaceAll("```json\\s*", "")
                .replaceAll("```\\s*", "")
                .trim();

        // Find the first { and last }
        int startIdx = cleaned.indexOf('{');
        int endIdx = cleaned.lastIndexOf('}');

        if (startIdx != -1 && endIdx != -1 && endIdx > startIdx) {
            return cleaned.substring(startIdx, endIdx + 1);
        }

        logger.warning("Could not find JSON object in response: " + text.substring(0, Math.min(200, text.length())));
        return "{}";
    }

            Do NOT include markdown code fences. Start your final response directly with { and end with }

            The JSON object must have these exact fields:
            {
                "title": "A clean, accurate headline",
                "article": "The main article content - EACH PARAGRAPH MUST BE WRAPPED WITH <p> TAGS",
                "summary": "A concise summary (2-3 sentences) of verified facts",
                "slug": "url-friendly-version-of-title (lowercase, hyphens, no special characters)",
                "sources": ["https://actual-source1.com", "https://actual-source2.com"]
            }

            CRITICAL: The article field MUST have each paragraph wrapped in <p></p> tags. Example:
            "<p>First paragraph with source attribution.</p><p>Second paragraph with quotes.</p>"

            Your lens: You are a local reporter for the Laudium community. Where appropriate, add context
            about Laudium or surrounding areas. Our main readers are the Muslim community - write from
            their perspective while maintaining strict journalistic accuracy.
            """;

        // Use chatWithTools to enable search_web and scrape_webpage with interleaved thinking
        String response = chatWithTools(systemPrompt, prompt, 3);
        logger.info("MiniMax article response received, length: " + response.length());

        // Parse JSON response
        try {
            String jsonContent = extractJsonObject(response);
            logger.fine("Extracted JSON: " + jsonContent.substring(0, Math.min(500, jsonContent.length())));

            ArticleGenerated article = gson.fromJson(jsonContent, ArticleGenerated.class);

            // Generate slug if not provided
            if (article.getSlug() == null || article.getSlug().isEmpty()) {
                article.setSlug(generateSlug(article.getTitle()));
            }

            logger.info("Successfully generated article: " + article.getTitle());
            return article;
        } catch (Exception e) {
            logger.warning("Failed to parse article JSON, returning raw content: " + e.getMessage());
            return new ArticleGenerated("Generated Article", response, "", generateSlug("Generated Article"));
        }
    }

    /**
     * Generate URL-friendly slug from title
     */
    private String generateSlug(String title) {
        if (title == null || title.isEmpty()) {
            return "generated-article";
        }
        return title.toLowerCase()
                .replaceAll("\\s+", "-")
                .replaceAll("[^a-z0-9-]", "")
                .replaceAll("-+", "-")
                .replaceAll("^-|-$", "");
    }

    /**
     * Extract JSON array from response text, handling various formats the model might return.
     * Looks for JSON array starting with [ and ending with ]
     *
     * @ModifiedByJunie 2025-12-10
     */
    private String extractJsonArray(String text) {
        if (text == null || text.isEmpty()) {
            return "[]";
        }

        // Remove markdown code fences
        String cleaned = text
                .replaceAll("```json\\s*", "")
                .replaceAll("```\\s*", "")
                .trim();

        // Find the first [ and last ]
        int startIdx = cleaned.indexOf('[');
        int endIdx = cleaned.lastIndexOf(']');

        if (startIdx != -1 && endIdx != -1 && endIdx > startIdx) {
            return cleaned.substring(startIdx, endIdx + 1);
        }

        // Try to find JSON array using regex
        Pattern jsonArrayPattern = Pattern.compile("\\[\\s*\\{.*}\\s*]", Pattern.DOTALL);
        Matcher matcher = jsonArrayPattern.matcher(cleaned);
        if (matcher.find()) {
            return matcher.group();
        }

        logger.warning("Could not find JSON array in response: " + text.substring(0, Math.min(200, text.length())));
        return "[]";
    }

    @Override
    public List<SourceArticle> sourceAll() throws IOException {

        String today = LocalDate.now().format(DateTimeFormatter.ofPattern("yyyy-MM-dd"));

    /**
     * Find sources for an article based on its title, summary, and/or content.
     * Uses MiniMax M2 to suggest relevant source URLs that could support the article.
     *
     * @param title The article title
     * @param summary The article summary (optional)
     * @param content The article content (optional, will be truncated if too long)
     * @return List of source URLs
     * @throws IOException if API call fails
     *
     * @ModifiedByJunie 2025-12-10
     */
    public List<String> findSources(String title, String summary, String content) throws IOException {
        logger.info("Finding sources for article: " + title);

        // Truncate content if too long to avoid token limits
        String truncatedContent = "";
        if (content != null && !content.isEmpty()) {
            // Strip HTML tags and truncate
            String plainContent = content.replaceAll("<[^>]*>", " ").replaceAll("\\s+", " ").trim();
            truncatedContent = plainContent.length() > 2000
                ? plainContent.substring(0, 2000) + "..."
                : plainContent;
        }

        String systemPrompt = """
            You are a research assistant specialized in finding credible news sources.
            Based on the article information provided, suggest 3-5 credible source URLs that would support this article.

            Focus on:
            - Major news outlets (Reuters, AP, Al Jazeera, BBC, etc.)
            - Official government or organization websites
            - Reputable regional news sources for South African content
            - Academic or research institutions when relevant

            IMPORTANT: Output ONLY a valid JSON array of URLs with NO text before or after.
            Do NOT include markdown code fences. Start directly with [ and end with ]

            Example format:
            ["https://www.reuters.com/article/...", "https://www.aljazeera.com/news/..."]

            Only include URLs that are likely to be real and relevant to the topic.
            If the topic is very specific or local, you may include fewer sources.
            """;

        StringBuilder userPrompt = new StringBuilder();
        userPrompt.append("Find credible sources for this article:\n\n");
        userPrompt.append("Title: ").append(title).append("\n");
        if (summary != null && !summary.isEmpty()) {
            userPrompt.append("Summary: ").append(summary).append("\n");
        }
        if (!truncatedContent.isEmpty()) {
            userPrompt.append("Content excerpt: ").append(truncatedContent).append("\n");
        }
        userPrompt.append("\nReturn ONLY a JSON array of source URLs.");

        String response = chatSimple(systemPrompt, userPrompt.toString());
        logger.info("MiniMax find-sources response received, length: " + response.length());

        try {
            String jsonContent = extractJsonArray(response);
            logger.fine("Extracted JSON: " + jsonContent);

            String[] sources = gson.fromJson(jsonContent, String[].class);

            if (sources == null || sources.length == 0) {
                logger.warning("No sources parsed from response");
                return List.of();
            }

            // Filter out obviously invalid URLs
            List<String> validSources = new ArrayList<>();
            for (String source : sources) {
                if (source != null && (source.startsWith("http://") || source.startsWith("https://"))) {
                    validSources.add(source.trim());
                }
            }

            logger.info("Successfully found " + validSources.size() + " sources");
            return validSources;
        } catch (Exception e) {
            logger.warning("Failed to parse sources JSON: " + e.getMessage());
            logger.fine("Raw response: " + response);
            return List.of();
        }
    }
